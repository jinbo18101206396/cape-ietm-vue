#### UT-003: 高度数值验证

**目的**: 验证高度计算正确

**验证脚本**:
```javascript
function validateHeight() {
  const editor = document.querySelector('.CodeMirror');
  const container = editor.closest('.source-pane');
  const viewport = window.innerHeight;
  
  const editorHeight = editor.offsetHeight;
  const containerHeight = container.offsetHeight;
  const ratio = (editorHeight / viewport * 100).toFixed(1);
  
  console.group('高度验证');
  console.log('CodeMirror高度:', editorHeight, 'px');
  console.log('容器高度:', containerHeight, 'px');
  console.log('Viewport高度:', viewport, 'px');
  console.log('高度比例:', ratio, '%');
  console.log('期望范围: 50% - 90%');
  
  const valid = ratio >= 50 && ratio <= 90;
  console.log('验证结果:', valid ? '✅ 正常' : '❌ 异常');
  console.groupEnd();
  
  return valid;
}

validateHeight();
```

#### IT-002: 压力测试

**测试目的**: 验证多次切换的稳定性

**步骤**:
```javascript
// 自动化压力测试脚本
async function stressTest() {
  const results = [];
  const iterations = 10;
  
  console.log('开始压力测试，共', iterations, '次切换');
  
  for (let i = 0; i < iterations; i++) {
    // 记录切换前高度
    const before = document.querySelector('.CodeMirror').offsetHeight;
    
    // 手动操作：进入设计视图 → 返回源码视图
    console.log(`第${i + 1}次切换，请手动操作后按回车继续...`);
    await new Promise(resolve => {
      const listener = (e) => {
        if (e.key === 'Enter') {
          window.removeEventListener('keydown', listener);
          resolve();
        }
      };
      window.addEventListener('keydown', listener);
    });
    
    // 等待DOM更新
    await new Promise(r => setTimeout(r, 500));
    
    // 记录切换后高度
    const after = document.querySelector('.CodeMirror').offsetHeight;
    const diff = Math.abs(after - before);
    
    results.push({
      iteration: i + 1,
      before,
      after,
      diff,
      valid: diff < 5
    });
    
    console.log(`第${i + 1}次: 前=${before}px, 后=${after}px, 差=${diff}px, ${diff < 5 ? '✅' : '❌'}`);
  }
  
  console.log('\n压力测试完成！');
  console.table(results);
  
  const passed = results.filter(r => r.valid).length;
  console.log(`通过: ${passed}/${iterations}, 通过率: ${(passed/iterations*100).toFixed(1)}%`);
}

// 运行测试
stressTest();
```

```javascript
// 测量视图切换性能
async function measureSwitchPerformance() {
  console.log('开始性能测试...');
  
  // 测量进入设计视图
  const t1 = performance.now();
  // 手动点击铅笔图标
  console.log('请点击铅笔图标...');
  
  // 等待用户操作
  await new Promise(resolve => setTimeout(resolve, 3000));
  const t2 = performance.now();
  console.log('进入设计视图耗时:', (t2 - t1).toFixed(2), 'ms');
  
  // 测量返回源码视图
  const t3 = performance.now();
  // 手动点击源码视图页签
  console.log('请点击源码视图页签...');
  
  await new Promise(resolve => setTimeout(resolve, 3000));
  const t4 = performance.now();
  console.log('返回源码视图耗时:', (t4 - t3).toFixed(2), 'ms');
  
  // 测量CodeMirror.refresh()
  const editor = document.querySelector('.CodeMirror');
  if (editor && editor.CodeMirror) {
    const t5 = performance.now();
    editor.CodeMirror.refresh();
    const t6 = performance.now();
    console.log('CodeMirror.refresh()耗时:', (t6 - t5).toFixed(2), 'ms');
  }
}

```

```
2. Ctrl+F5 → 强制刷新
```
