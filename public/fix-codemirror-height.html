CodeMirror 高度问题手动修复测试

步骤：

  1. 在 DM 编辑器页面按 F12 打开控制台
  2. 双击 para 进入设计视图，点击"保存"返回
  3. 复制下面的代码到控制台运行

const wrapper = document.querySelector('.CodeMirror');
const cm = wrapper.CodeMirror;
const scroller = cm.display.scroller;
const sizer = cm.display.sizer;

手动修复代码

// 手动修复 CodeMirror 高度
const wrapper = document.querySelector('.CodeMirror');
const cm = wrapper.CodeMirror;
const scroller = cm.display.scroller;

console.log('=== 修复前 ===');
console.log('空白区域:', wrapper.offsetHeight - scroller.offsetHeight, 'px');

// 方案1：清除 scroller 的固定高度
scroller.style.height = '';
scroller.style.minHeight = '';
scroller.style.maxHeight = '';

// 方案2：强制 refresh 多次
cm.refresh();
requestAnimationFrame(() => {
  cm.refresh();
  setTimeout(() => {
    cm.refresh();

    console.log('=== 修复后 ===');
    console.log('空白区域:', wrapper.offsetHeight - scroller.offsetHeight, 'px');

    if (wrapper.offsetHeight - scroller.offsetHeight < 100) {
      console.log('%c✅ 修复成功！', 'color: green; font-weight: bold');
    } else {
      console.log('%c❌ 修复失败', 'color: red; font-weight: bold');
    }
  }, 100);
});

强制高度修复代码（如果上面不行）

// 强制设置 scroller 高度为 wrapper 高度
const wrapper = document.querySelector('.CodeMirror');
const cm = wrapper.CodeMirror;
const scroller = cm.display.scroller;

console.log('=== 强制修复 ===');
console.log('修复前空白:', wrapper.offsetHeight - scroller.offsetHeight, 'px');

// 清除所有高度样式
scroller.style.height = '';
scroller.style.minHeight = '';
scroller.style.maxHeight = '';

// 设置为自动高度
scroller.style.height = 'auto';

// 刷新
cm.refresh();

setTimeout(() => {
  const newBlank = wrapper.offsetHeight - scroller.offsetHeight;
  console.log('修复后空白:', newBlank, 'px');

  // 如果还是有问题，强制设置高度
  if (Math.abs(newBlank) > 50) {
    console.log('使用强制高度方案...');
    scroller.style.height = wrapper.clientHeight + 'px';
    cm.refresh();

    setTimeout(() => {
      console.log('最终空白:', wrapper.offsetHeight - scroller.offsetHeight, 'px');
    }, 100);
  }
}, 100);
