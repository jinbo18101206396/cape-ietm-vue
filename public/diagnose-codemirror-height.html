CodeMirror 高度诊断工具

使用说明：

  1. 在浏览器中打开 DM 编辑器页面
  2. 按 F12 打开开发者工具控制台
  3. 复制下面的诊断代码到控制台运行

步骤1：初始状态诊断（刚进入编辑器）

// 初始状态诊断
(function() {
  const wrapper = document.querySelector('.CodeMirror');
  const scroller = document.querySelector('.CodeMirror-scroll');
  const sizer = document.querySelector('.CodeMirror-sizer');
  const gutters = document.querySelector('.CodeMirror-gutters');

  if (!wrapper || !scroller) {
    console.error('❌ CodeMirror 未找到！请确保在编辑器页面运行此代码。');
    return;
  }

  const diagnostics = {
    wrapper: {
      offsetHeight: wrapper.offsetHeight,
      clientHeight: wrapper.clientHeight,
      styleHeight: wrapper.style.height,
      computedHeight: getComputedStyle(wrapper).height
    },
    scroller: {
      offsetHeight: scroller.offsetHeight,
      styleHeight: scroller.style.height,
      computedHeight: getComputedStyle(scroller).height
    },
    sizer: {
      offsetHeight: sizer?.offsetHeight || 0
    },
    gutters: {
      offsetWidth: gutters?.offsetWidth || 0
    },
    blank: wrapper.offsetHeight - scroller.offsetHeight
  };

  console.log('=== 【初始状态】CodeMirror 布局诊断 ===');
  console.log('Wrapper 高度:', diagnostics.wrapper.offsetHeight, 'px');
  console.log('Scroller 高度:', diagnostics.scroller.offsetHeight, 'px');
  console.log('空白区域:', diagnostics.blank, 'px');
  console.log('Gutters 宽度:', diagnostics.gutters.offsetWidth, 'px');
  console.log('');
  console.log('完整诊断数据:', diagnostics);

  // 判断
  if (diagnostics.blank > 100) {
    console.error('❌ 检测到异常：空白区域 ' + diagnostics.blank + 'px > 100px');
    console.error('   说明 scroller 高度异常，需要修复！');
  } else {
    console.log('%c✅ 初始状态正常：空白区域 < 100px', 'color: green; font-weight: bold');
  }

  // 保存到全局变量供后续对比
  window.__cmInitialDiagnostics = diagnostics;
})();

步骤2：进入设计视图后返回诊断

操作：双击任意 para 节点 → 进入设计视图 → 点击"保存"返回源码视图 → 运行下面的代码 

// 从设计视图返回后诊断
(function() {
  const wrapper = document.querySelector('.CodeMirror');
  const scroller = document.querySelector('.CodeMirror-scroll');
  const sizer = document.querySelector('.CodeMirror-sizer');
  const gutters = document.querySelector('.CodeMirror-gutters');

  if (!wrapper || !scroller) {
    console.error('❌ CodeMirror 未找到！');
    return;
  }

  const diagnostics = {
    wrapper: {
      offsetHeight: wrapper.offsetHeight,
      clientHeight: wrapper.clientHeight,
      styleHeight: wrapper.style.height,
      computedHeight: getComputedStyle(wrapper).height
    },
    scroller: {
      offsetHeight: scroller.offsetHeight,
      styleHeight: scroller.style.height,
      computedHeight: getComputedStyle(scroller).height
    },
    sizer: {
      offsetHeight: sizer?.offsetHeight || 0
    },
    gutters: {
      offsetWidth: gutters?.offsetWidth || 0
    },
    blank: wrapper.offsetHeight - scroller.offsetHeight
  };

  console.log('=== 【从设计视图返回后】CodeMirror 布局诊断 ===');
  console.log('Wrapper 高度:', diagnostics.wrapper.offsetHeight, 'px');
  console.log('Scroller 高度:', diagnostics.scroller.offsetHeight, 'px');
  console.log('空白区域:', diagnostics.blank, 'px');
  console.log('Gutters 宽度:', diagnostics.gutters.offsetWidth, 'px');
  console.log('');
  console.log('完整诊断数据:', diagnostics);

  // 判断
  if (diagnostics.blank > 100) {
    console.error('❌ 检测到异常：空白区域 ' + diagnostics.blank + 'px > 100px');
    console.error('   说明修复失败，scroller 高度仍然异常！');

    // 检查 setSize 是否被调用
    if (wrapper.style.width !== '100%' || wrapper.style.height !== '100%') {
      console.error('   wrapper.style 未设置为 100%，说明 setSize() 未生效！');
      console.error('   wrapper.style.width =', wrapper.style.width);
      console.error('   wrapper.style.height =', wrapper.style.height);
    }

    // 检查 scroller.style.height 是否被强制设置
    if (!scroller.style.height || scroller.style.height === '') {
      console.error('   scroller.style.height 未设置，说明修复代码未执行！');
    }
  } else {
    console.log('%c✅ 修复成功：空白区域 < 100px', 'color: green; font-weight: bold');
  }

  // 对比初始状态
  if (window.__cmInitialDiagnostics) {
    const initial = window.__cmInitialDiagnostics;
    console.log('');
    console.log('=== 对比初始状态 ===');
    console.log('Wrapper 高度变化:', diagnostics.wrapper.offsetHeight - initial.wrapper.offsetHeight, 'px');
    console.log('Scroller 高度变化:', diagnostics.scroller.offsetHeight - initial.scroller.offsetHeight, 'px');
    console.log('空白区域变化:', diagnostics.blank - initial.blank, 'px');

    if (Math.abs(diagnostics.blank - initial.blank) > 50) {
      console.warn('⚠️ 警告：空白区域变化超过 50px，说明布局发生了异常变化');
    }
  }
})();

步骤3：手动修复验证

如果步骤2发现空白区域 > 100px，运行下面的代码手动修复，验证修复逻辑是否正确 

// 手动修复 scroller 高度
(function() {
  const wrapper = document.querySelector('.CodeMirror');
  const scroller = document.querySelector('.CodeMirror-scroll');

  if (!wrapper || !scroller) {
    console.error('❌ CodeMirror 未找到！');
    return;
  }

  console.log('=== 手动修复 scroller 高度 ===');

  const beforeBlank = wrapper.offsetHeight - scroller.offsetHeight;
  console.log('修复前：空白区域 =', beforeBlank, 'px');

  // 修复1：强制设置 scroller 高度
  scroller.style.height = wrapper.offsetHeight + 'px';
  console.log('步骤1：设置 scroller.style.height =', wrapper.offsetHeight, 'px');

  // 修复2：调用 CodeMirror 的 setSize()
  const cm = wrapper.CodeMirror;
  if (cm) {
    cm.setSize(null, null);
    cm.setSize('100%', '100%');
    console.log('步骤2：调用 cm.setSize(null, null) 和 cm.setSize("100%", "100%")');
  } else {
    console.warn('⚠️ 警告：wrapper.CodeMirror 未找到，无法调用 setSize()');
  }

  // 检查修复效果
  setTimeout(() => {
    const afterBlank = wrapper.offsetHeight - scroller.offsetHeight;
    console.log('修复后：空白区域 =', afterBlank, 'px');

    if (afterBlank < 100) {
      console.log('%c✅ 手动修复成功！', 'color: green; font-weight: bold');
      console.log('   说明修复逻辑正确，但可能 forceFixGuttersLayout() 未被调用');
    } else {
      console.error('❌ 手动修复失败！');
      console.error('   说明修复逻辑本身存在问题');
    }
  }, 100);
})();
